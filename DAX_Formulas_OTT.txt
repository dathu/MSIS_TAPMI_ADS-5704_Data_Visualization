5. Date:

dim_date = 
ADDCOLUMNS (
    CALENDAR (DATE(2024, 1, 1), DATE(2024, 11, 30)),
    "Year", YEAR([Date]),
    "Month", FORMAT([Date], "MMMM"),
    "MonthNum", MONTH([Date]),
    "Quarter", "Q" & FORMAT([Date], "Q"),
    "Weekday", FORMAT([Date], "dddd"),
    "WeekdayNum", WEEKDAY([Date], 2)  -- Monday = 1, Sunday = 7
)

Adding New cloumn to subscriber table:

UserStatus = 
SWITCH(
    TRUE(),
    ISBLANK('subscribers'[last_active_date]), "Active",
    'subscribers'[last_active_date] < TODAY() - 30, "Inactive"
)


5. Active User Count

Active User Count =
CALCULATE(
    DISTINCTCOUNT(subscribers[user_id]),
    subscribers[UserStatus] = "Active"
)


*****************************

6. Inactive User Count

Inactive User Count = 
CALCULATE(
    DISTINCTCOUNT(subscribers[user_id]),
    subscribers[UserStatus] = "Inactive"
